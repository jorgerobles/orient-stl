**web/src/viewport/SupportRenderer.ts** — New class:

```typescript
import * as THREE from 'three';
import type { SupportResult, Support, RaftGeometry } from '../types';

const TYPE_COLORS = {
  light: 0x4caf50,   // green
  medium: 0xffc107,  // yellow/amber
  heavy: 0xf44336,   // red
};

export class SupportRenderer {
  private scene: THREE.Scene;
  private group: THREE.Group;  // all support geometry in one group for easy toggle
  private columnMeshes: THREE.Mesh[] = [];
  private raftMesh: THREE.Mesh | null = null;
  private raftLines: THREE.LineSegments | null = null;

  constructor(scene: THREE.Scene) {
    this.scene = scene;
    this.group = new THREE.Group();
    this.group.name = 'supports';
    this.group.visible = false;
    this.scene.add(this.group);
  }

  render(result: SupportResult) {
    this.clear();

    // Render raft first (base layer)
    this.renderRaft(result.raft);

    // Render each support column
    for (const support of result.supports) {
      this.renderColumn(support);
    }
  }

  private renderColumn(support: Support) {
    const { contact } = support;
    const base = new THREE.Vector3(...contact.base);
    const tip = new THREE.Vector3(...contact.position);
    const height = base.distanceTo(tip);
    const radius = contact.tipDiameter / 2;

    const geo = new THREE.CylinderGeometry(radius, radius, height, 8);
    const mat = new THREE.MeshBasicMaterial({
      color: TYPE_COLORS[contact.supportType],
      transparent: true,
      opacity: 0.7,
    });
    const mesh = new THREE.Mesh(geo, mat);

    // Position at midpoint between base and tip
    const mid = base.clone().add(tip).multiplyScalar(0.5);
    mesh.position.copy(mid);

    // Orient cylinder along the support direction
    const dir = tip.clone().sub(base).normalize();
    const up = new THREE.Vector3(0, 1, 0);
    const quat = new THREE.Quaternion().setFromUnitVectors(up, dir);
    mesh.quaternion.copy(quat);

    this.group.add(mesh);
    this.columnMeshes.push(mesh);
  }

  private renderRaft(raft: RaftGeometry) {
    if (raft.vertices.length === 0) return;

    // Raft mesh (triangles)
    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(raft.vertices, 3));
    geo.setIndex(new THREE.BufferAttribute(raft.triangles, 1));
    geo.computeVertexNormals();

    const mat = new THREE.MeshBasicMaterial({
      color: 0x9e9e9e,
      transparent: true,
      opacity: 0.3,
      side: THREE.DoubleSide,
    });
    this.raftMesh = new THREE.Mesh(geo, mat);
    this.group.add(this.raftMesh);

    // Raft line connections
    if (raft.lines.length > 0) {
      const lineGeo = new THREE.BufferGeometry();
      lineGeo.setAttribute('position', new THREE.Float32BufferAttribute(raft.vertices, 3));
      lineGeo.setIndex(new THREE.BufferAttribute(new Uint32Array(raft.lines), 1));

      const lineMat = new THREE.LineBasicMaterial({ color: 0x616161, linewidth: 2 });
      this.raftLines = new THREE.LineSegments(lineGeo, lineMat);
      this.group.add(this.raftLines);
    }
  }

  clear() {
    this.columnMeshes.forEach(m => {
      m.geometry.dispose();
      (m.material as THREE.Material).dispose();
      this.group.remove(m);
    });
    this.columnMeshes = [];

    if (this.raftMesh) {
      this.raftMesh.geometry.dispose();
      (this.raftMesh.material as THREE.Material).dispose();
      this.group.remove(this.raftMesh);
      this.raftMesh = null;
    }
    if (this.raftLines) {
      this.raftLines.geometry.dispose();
      (this.raftLines.material as THREE.Material).dispose();
      this.group.remove(this.raftLines);
      this.raftLines = null;
    }
  }

  setVisible(visible: boolean) {
    this.group.visible = visible;
  }

  dispose() {
    this.clear();
    this.scene.remove(this.group);
  }
}
```
