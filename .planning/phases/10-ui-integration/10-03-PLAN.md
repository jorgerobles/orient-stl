function writeBinarySTL(positions: number[], normals: number[], filename: string) {
  const triangleCount = positions.length / 9;
  const buffer = new ArrayBuffer(84 + triangleCount * 50); // 80 header + 4 count + 50 per tri
  const view = new DataView(buffer);

  // Header (80 bytes)
  const header = `orient-stl support export`;
  for (let i = 0; i < 80; i++) {
    view.setUint8(i, i < header.length ? header.charCodeAt(i) : 0);
  }

  // Triangle count
  view.setUint32(80, triangleCount, true);

  // Triangles
  let offset = 84;
  for (let i = 0; i < triangleCount; i++) {
    const base = i * 9;
    // Normal (3 floats)
    view.setFloat32(offset, normals[base], true); offset += 4;
    view.setFloat32(offset, normals[base+1], true); offset += 4;
    view.setFloat32(offset, normals[base+2], true); offset += 4;
    // Vertex 1
    view.setFloat32(offset, positions[base], true); offset += 4;
    view.setFloat32(offset, positions[base+1], true); offset += 4;
    view.setFloat32(offset, positions[base+2], true); offset += 4;
    // Vertex 2
    view.setFloat32(offset, positions[base+3], true); offset += 4;
    view.setFloat32(offset, positions[base+4], true); offset += 4;
    view.setFloat32(offset, positions[base+5], true); offset += 4;
    // Vertex 3
    view.setFloat32(offset, positions[base+6], true); offset += 4;
    view.setFloat32(offset, positions[base+7], true); offset += 4;
    view.setFloat32(offset, positions[base+8], true); offset += 4;
    // Attribute byte count
    view.setUint16(offset, 0, true); offset += 2;
  }

  const blob = new Blob([buffer], { type: 'application/octet-stream' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
}
