**web/src/views/SupportPanel.ts** — New UI component:

```typescript
export class SupportPanel {
  private root: HTMLElement;
  private enabled = false;
  private config: SupportConfig = defaultSupportConfig();

  constructor(container: HTMLElement) {
    this.root = document.createElement('section');
    this.root.className = 'support-panel';
    this.root.innerHTML = `
      <h3>Support Generation</h3>
      <label class="toggle">
        <input type="checkbox" id="support-toggle" />
        <span>Enable supports</span>
      </label>
      <div class="support-config" hidden>
        <label>Layer height: <input type="number" id="support-layer" value="0.05" step="0.01" min="0.01" max="0.1" /> mm</label>
        <label>Light threshold: <input type="number" id="support-light" value="50" step="10" /> mm³</label>
        <label>Medium threshold: <input type="number" id="support-medium" value="500" step="50" /> mm³</label>
        <label>Light tip: <input type="number" id="support-light-tip" value="0.25" step="0.05" /> mm</label>
        <label>Medium tip: <input type="number" id="support-medium-tip" value="0.40" step="0.05" /> mm</label>
        <label>Heavy tip: <input type="number" id="support-heavy-tip" value="0.80" step="0.05" /> mm</label>
      </div>
    `;
    container.appendChild(this.root);
    this.bindEvents();
  }

  private bindEvents() {
    const toggle = this.root.querySelector('#support-toggle') as HTMLInputElement;
    const configDiv = this.root.querySelector('.support-config') as HTMLDivElement;

    toggle.addEventListener('change', () => {
      this.enabled = toggle.checked;
      configDiv.hidden = !this.enabled;
      this.emit();
    });

    // Config inputs emit on change
    this.root.querySelectorAll('.support-config input').forEach(input => {
      input.addEventListener('change', () => this.emit());
    });
  }

  private emit() {
    // Read all config values from DOM, dispatch CustomEvent
    const event = new CustomEvent('support-config-change', {
      detail: { enabled: this.enabled, config: this.readConfig() }
    });
    this.root.dispatchEvent(event);
  }

  private readConfig(): SupportConfig {
    return {
      layerHeight: parseFloat((this.root.querySelector('#support-layer') as HTMLInputElement).value),
      lightThreshold: parseFloat((this.root.querySelector('#support-light') as HTMLInputElement).value),
      mediumThreshold: parseFloat((this.root.querySelector('#support-medium') as HTMLInputElement).value),
      // ... read all fields
    };
  }
}
```
