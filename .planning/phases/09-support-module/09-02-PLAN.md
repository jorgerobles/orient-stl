```rust
#[cfg(feature = "wasm")]
#[wasm_bindgen]
pub fn generate_supports(
    positions: &[f32],
    normals: &[f32],
    areas: &[f32],
    direction: &[f32],
    config: JsValue,
) -> JsValue {
    let config: SupportConfig = serde_wasm_bindgen::from_value(config)
        .unwrap_or_else(|e| wasm_bindgen::throw_str(&format!("Invalid config: {e}")));

    // 1. Detect islands
    let islands = island::detect_islands(positions, normals, direction, &config);

    // 2. For each island: classify volume, place contacts
    let mut all_contacts = Vec::new();
    for island in &islands {
        let centroid_3d = project_to_mesh(island, positions, normals, direction);
        let vol = volume::compute_volume_above(positions, normals, areas, &island.centroid, direction);
        let stype = volume::classify_support_type(vol, &config);
        let contacts = placement::place_contacts(island, positions, normals, direction, &stype, &config);
        for mut c in contacts {
            c.volume_above = vol;  // store for result
            all_contacts.push(c);
        }
    }

    // 3. Generate raft
    let raft = raft::generate_raft(&all_contacts, &config);

    // 4. Build supports from contacts
    let supports: Vec<Support> = all_contacts.iter().map(|c| Support {
        contact: c.clone(),
        volume_above: c.volume_above,
    }).collect();

    let result = SupportResult {
        supports,
        raft,
        total_volume: /* sum of volumes */,
        island_count: islands.len() as u32,
    };

    serde_wasm_bindgen::to_value(&result).unwrap()
}
```
